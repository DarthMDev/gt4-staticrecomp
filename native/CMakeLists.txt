cmake_minimum_required(VERSION 3.24)
project(GT4NativeVulkan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

get_filename_component(GT4_ROOT ".." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(GT4_ASSETS_ROOT "${GT4_ROOT}" CACHE PATH "Verified private assets checkout")
set(GT4_EE_PROFILE "production" CACHE STRING "Pinned EE library profile")
set_property(CACHE GT4_EE_PROFILE PROPERTY STRINGS production baseline recovered observation)
set(GT4_SHAPE_RUNTIME_SOURCE "" CACHE PATH "Verified experimental runtime overlay capsule")
set(DONOR "${GT4_ASSETS_ROOT}/private/references/ps2-web-recomp")
set(PORT "${GT4_ROOT}/private/slice114-native-base/port")
set(SHADERS "${GT4_ROOT}/private/slice114-native-base/shaders")
set(VULKAN_HEADERS "${GT4_ASSETS_ROOT}/private/tooling/pcsx2-slice101-source/3rdparty/vulkan/include")
if(GT4_EE_PROFILE STREQUAL "production")
    set(GAME_LIBRARY "${GT4_ASSETS_ROOT}/private/reference-acceleration-study/runtime-build/ps2game.lib")
    set(expected_game_library_hash "c837d13dae027f377aecdb0c39834ede2e9463eabf7f767fda6a3c5a0762de1f")
elseif(GT4_EE_PROFILE STREQUAL "baseline" OR GT4_EE_PROFILE STREQUAL "recovered"
       OR GT4_EE_PROFILE STREQUAL "observation")
    file(READ "${GT4_ROOT}/docs/inputs/slice118-ee-libraries.json" candidate_manifest)
    if(GT4_EE_PROFILE STREQUAL "observation")
        if(NOT GT4_SHAPE_RUNTIME_SOURCE)
            message(FATAL_ERROR "Observation profile requires verified runtime overlays")
        endif()
        set(observation_library_manifest "${GT4_ROOT}/docs/inputs/slice131-observation-library.json")
        file(READ "${observation_library_manifest}" observation_library)
        string(JSON library_path GET "${observation_library}" profile path)
        string(JSON expected_game_library_hash GET "${observation_library}" profile sha256)
        if(NOT library_path STREQUAL "private/slice131-observation-library/game-build/ps2game.lib")
            message(FATAL_ERROR "Observation library must be the qualified slice 131 capsule")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${observation_library_manifest}")
    else()
        string(JSON library_path GET "${candidate_manifest}" profiles "${GT4_EE_PROFILE}" path)
        string(JSON expected_game_library_hash GET "${candidate_manifest}" profiles "${GT4_EE_PROFILE}" sha256)
        if(NOT library_path MATCHES "^private/slice118-ee-candidates/" OR library_path MATCHES "\\.\\.")
            message(FATAL_ERROR "Candidate library must stay within the isolated slice 118 capsule")
        endif()
    endif()
    string(JSON port_path GET "${candidate_manifest}" port_directory)
    string(JSON shader_path GET "${candidate_manifest}" shader_directory)
    foreach(candidate_path IN ITEMS "${port_path}" "${shader_path}")
        if(NOT candidate_path MATCHES "^private/slice118-ee-candidates/" OR candidate_path MATCHES "\\.\\.")
            message(FATAL_ERROR "Candidate paths must stay within the isolated slice 118 capsule")
        endif()
    endforeach()
    set(GAME_LIBRARY "${GT4_ROOT}/${library_path}")
    set(PORT "${GT4_ROOT}/${port_path}")
    set(SHADERS "${GT4_ROOT}/${shader_path}")
else()
    message(FATAL_ERROR "Unknown GT4_EE_PROFILE; no permissive AOT library selection")
endif()
if(NOT EXISTS "${GAME_LIBRARY}")
    message(FATAL_ERROR "Missing private AOT game library; see docs/BUILD.md")
endif()
file(SHA256 "${GAME_LIBRARY}" game_library_hash)
if(NOT game_library_hash STREQUAL expected_game_library_hash)
    message(FATAL_ERROR "AOT game library differs from the qualified USA v2.00 producer")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${GAME_LIBRARY}")
set(RUNTIME_SOURCES memory dispatch system kernel hw cop disc dmac vif vu_micro vu_jit vu_jit_web)
set(RUNTIME_FILES)
foreach(source IN LISTS RUNTIME_SOURCES)
    list(APPEND RUNTIME_FILES "${DONOR}/runtime/src/${source}.cpp")
endforeach()
foreach(source IN ITEMS ipu/ipu gs/gif gs/queue gs/raster
                        iop/cpu iop/kernel iop/libs iop/hw iop/spu2 iop/sio2
                        iop/io iop/cdvd iop/sif iop/cdvdfsv)
    list(APPEND RUNTIME_FILES "${DONOR}/runtime/src/${source}.cpp")
endforeach()
add_library(ps2runtime_vulkan STATIC ${RUNTIME_FILES} "${PORT}/gs.cpp" "${PORT}/gpu.cpp" native_vulkan.cpp)
target_include_directories(ps2runtime_vulkan PUBLIC "${DONOR}/runtime/include" "${DONOR}/runtime/src"
                           "${DONOR}/runtime/src/gs" "${CMAKE_CURRENT_SOURCE_DIR}" "${VULKAN_HEADERS}")
target_compile_definitions(ps2runtime_vulkan PUBLIC PS2_GPU _CRT_SECURE_NO_WARNINGS
                           GT4_VULKAN_SHADER_DIR="${SHADERS}")
target_compile_options(ps2runtime_vulkan PRIVATE /W4)
add_library(native_host_io STATIC host_io.cpp)
target_link_libraries(native_host_io PUBLIC winmm xinput user32)
target_compile_options(native_host_io PRIVATE /W4)
add_executable(gt4vulkan "${PORT}/main.cpp" window.cpp)
target_include_directories(gt4vulkan PRIVATE "${PORT}")
target_compile_definitions(gt4vulkan PRIVATE PS2_WINDOW)
target_link_libraries(gt4vulkan PRIVATE
    "${GAME_LIBRARY}"
    ps2runtime_vulkan native_host_io user32)
add_executable(vulkan_fixture fixture.cpp)
target_include_directories(vulkan_fixture PRIVATE "${PORT}")
target_link_libraries(vulkan_fixture PRIVATE ps2runtime_vulkan user32)
target_compile_options(gt4vulkan PRIVATE /W4)
target_compile_options(vulkan_fixture PRIVATE /W4)
add_executable(host_io_fixture host_io_fixture.cpp)
target_link_libraries(host_io_fixture PRIVATE native_host_io)
target_compile_options(host_io_fixture PRIVATE /W4)
add_executable(shape_observer_fixture shape_observer_fixture.cpp shape_observer.cpp)
target_compile_options(shape_observer_fixture PRIVATE /W4)
add_executable(perf_accounting_fixture perf_accounting_fixture.cpp)
target_compile_options(perf_accounting_fixture PRIVATE /W4)
add_executable(shape_capture_export_fixture shape_capture_export_fixture.cpp shape_capture_export.cpp shape_observer.cpp)
target_compile_options(shape_capture_export_fixture PRIVATE /W4)
add_executable(shape_capture_control_fixture shape_capture_control_fixture.cpp shape_capture_control.cpp shape_capture_export.cpp shape_observer.cpp)
target_compile_options(shape_capture_control_fixture PRIVATE /W4)
if(GT4_SHAPE_RUNTIME_SOURCE)
    get_filename_component(observation_source "${GT4_SHAPE_RUNTIME_SOURCE}" ABSOLUTE)
    file(RELATIVE_PATH observation_relative "${GT4_ROOT}" "${observation_source}")
    if(NOT observation_relative MATCHES "^private/[^/]+$" OR observation_relative MATCHES "\\.\\.")
        message(FATAL_ERROR "Observation sources must be an isolated private capsule")
    endif()
    file(READ "${GT4_ROOT}/docs/inputs/slice130-shape-runtime.json" observation_manifest)
    file(GLOB observation_files CONFIGURE_DEPENDS RELATIVE "${observation_source}" "${observation_source}/*")
    list(SORT observation_files)
    if(NOT observation_files STREQUAL "dispatch.cpp;kernel.cpp;runtime-overlay.json;system.cpp")
        message(FATAL_ERROR "Observation capsule must contain only the three sources and receipt")
    endif()
    set(observation_overlays)
    foreach(source IN ITEMS dispatch.cpp kernel.cpp system.cpp)
        string(JSON expected_overlay_hash GET "${observation_manifest}" runtime_source_sha256 "${source}")
        file(SHA256 "${observation_source}/${source}" actual_overlay_hash)
        if(NOT actual_overlay_hash STREQUAL expected_overlay_hash)
            message(FATAL_ERROR "Observation overlay differs from reviewed source")
        endif()
        list(APPEND observation_overlays "${observation_source}/${source}")
    endforeach()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${observation_overlays}
        "${observation_source}/runtime-overlay.json" "${GT4_ROOT}/docs/inputs/slice130-shape-runtime.json")
    if(GT4_EE_PROFILE STREQUAL "observation")
        target_sources(gt4vulkan PRIVATE ${observation_overlays} shape_observer.cpp shape_observer_adapter.cpp
                       shape_capture_control.cpp shape_capture_export.cpp)
        target_compile_definitions(gt4vulkan PRIVATE GT4_SHAPE_CAPTURE)
        target_link_options(gt4vulkan PRIVATE "/MAP:${CMAKE_BINARY_DIR}/observation-game.map" /VERBOSE:LIB)
    endif()
    foreach(fixture IN ITEMS shape_runtime_fixture shape_runtime_control)
        add_executable(${fixture} shape_runtime_fixture.cpp shape_observer.cpp shape_observer_adapter.cpp)
        target_link_libraries(${fixture} PRIVATE "${GAME_LIBRARY}" ps2runtime_vulkan user32)
        target_compile_options(${fixture} PRIVATE /W4)
    endforeach()
    target_sources(shape_runtime_fixture PRIVATE ${observation_overlays})
    target_compile_definitions(shape_runtime_control PRIVATE GT4_OBSERVER_CONTROL)
    target_link_options(shape_runtime_fixture PRIVATE "/MAP:${CMAKE_BINARY_DIR}/shape-runtime.map" /VERBOSE:LIB)
endif()
